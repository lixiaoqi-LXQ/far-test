cmake_minimum_required(VERSION 3.20)
project(raf VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(RAF_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(RAF_THREAD_SANITIZE "Enable thread sanitizer" OFF)
option(RAF_PORTABLE_CRC "Use only the portable CRC implementation" OFF)
add_library(raf STATIC src/raf.cpp src/crc32c.cpp)
target_include_directories(raf PUBLIC include)
target_link_libraries(raf PUBLIC Threads::Threads)
target_compile_options(raf PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(RAF_PORTABLE_CRC)
  target_compile_definitions(raf PRIVATE RAF_PORTABLE_CRC=1)
endif()
if(RAF_SANITIZE)
  target_compile_options(raf PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(raf PUBLIC -fsanitize=address,undefined)
endif()
if(RAF_THREAD_SANITIZE)
  target_compile_options(raf PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(raf PUBLIC -fsanitize=thread)
endif()
enable_testing()
add_executable(raf_tests tests/raf_tests.cpp tests/faults.cpp)
target_link_libraries(raf_tests PRIVATE raf)
target_compile_options(raf_tests PRIVATE -Wall -Wextra -Wpedantic)
foreach(call writev write pwrite pread fdatasync ftruncate)
  target_link_options(raf_tests PRIVATE "-Wl,--wrap=${call}")
endforeach()
add_test(NAME raf_tests COMMAND raf_tests)
set_tests_properties(raf_tests PROPERTIES TIMEOUT 120)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/bench/raf_bench.cpp")
  add_executable(raf_bench bench/raf_bench.cpp)
  target_link_libraries(raf_bench PRIVATE raf)
  target_compile_options(raf_bench PRIVATE -Wall -Wextra -Wpedantic)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME benchmark_smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/bench_smoke.py $<TARGET_FILE:raf_bench>)
  endif()
endif()
